03
模型下载脚本 - hf.sh
HuggingFace 模型快速下载工具

功能特性

✓支持 URL 或 repo/file 两种方式指定模型
✓自动识别 curl 或 wget 下载工具
✓自动替换 blob/main → resolve/main
✓支持自定义输出目录

使用示例

./scripts/hf.sh --repo TheBloke/Mixtral-8x7B \
                 --file mixtral-8x7b-v0.1.Q4_K_M.gguf

./llama-cli -m $(./scripts/hf.sh <URL>)
#!/usr/bin/env bash

# 检测可用的下载工具
if has_cmd wget; then
    cmd="wget -q -c -O %s/%s %s"
elif has_cmd curl; then
    cmd="curl -C - -f --output-dir %s -o %s -L %s"
else
    log "[E] curl or wget not found"
    exit 1
fi

# 解析参数
while [[ $# -gt 0 ]]; do
    case "$1" in
        --repo) repo="$2"; shift 2 ;;
        --file) file="$2"; shift 2 ;;
        --outdir) outdir="$2"; shift 2 ;;
    esac
done

# 替换 blob/main 为 resolve/main
url=${url/blob\/main/resolve\/main}

# 执行下载
cmd=$(printf "$cmd" "$outdir" "$basename" "$url")
$cmd

# bench-models.sh 核心逻辑
run_model() {
    local HFR=$1  # HuggingFace Repo
    local HFF=$2  # HuggingFace File

    # 运行 llama-batched-bench
    ./bin/llama-batched-bench \
        -hfr "${HFR}" -hff "${HFF}" \
        -m "${HFF}" -fa 1 -ub 2048 \
        ${ARGS_BB}

    # 运行 llama-bench
    ./bin/llama-bench \
        -m "${HFF}" -fa 1 -ub 2048 \
        ${ARGS_B}
}

# 测试多个模型
run_model "ggml-org/Qwen3-Coder-30B-A3B..."
run_model "ggml-org/Qwen2.5-Coder-7B..."
run_model "ggml-org/gemma-3-4b-it-qat..."

#!/usr/bin/env bash
# sync-ggml.sh - 同步 ggml 子模块

# 同步 CMake 配置
cp -rpv ../ggml/CMakeLists.txt ./ggml/
cp -rpv ../ggml/src/CMakeLists.txt ./ggml/src/

# 同步 cmake 文件
cp -rpv ../ggml/cmake/* ./ggml/cmake/

# 同步源文件
cp -rpv ../ggml/src/ggml* ./ggml/src/

# 同步头文件
cp -rpv ../ggml/include/ggml*.h ./ggml/include/
cp -rpv ../ggml/include/gguf*.h ./ggml/include/

# 同步测试文件
cp -rpv ../ggml/tests/test-*.cpp ./tests/

# 同步许可证和脚本
cp -rpv ../LICENSE ./
cp -rpv ../ggml/scripts/gen-authors.sh ./scripts/

// serve-static.js - Node.js 静态文件服务器
const http = require('http');
const fs = require('fs').promises;
const path = require('path');

const server = http.createServer(async (req, res) => {
  // 设置 CORS headers
  res.setHeader('Cross-Origin-Opener-Policy',
                'same-origin');
  res.setHeader('Cross-Origin-Embedder-Policy',
                'require-corp');

  const filePath = path.join(STATIC_DIR, req.url);
  const stats = await fs.stat(filePath);

  if (stats.isDirectory()) {
    // 查找 index.html 或生成目录列表
  } else {
    // 返回静态文件
    const data = await fs.readFile(filePath);
    res.end(data);
  }
});
